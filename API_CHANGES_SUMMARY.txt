# 2. SERIALIZERS.PY - Nuevos serializers

class ViaStatSerializer(serializers.Serializer):
    """Para por_via endpoint"""
    via = serializers.CharField()
    total_siniestros = serializers.IntegerField()
    total_lesionados = serializers.IntegerField()
    total_fallecidos = serializers.IntegerField()


class CausaProbableStatSerializer(serializers.Serializer):
    """Para por_causa_probable endpoint"""
    id = serializers.IntegerField()
    causa = serializers.CharField()
    total_siniestros = serializers.IntegerField()
    total_lesionados = serializers.IntegerField()
    total_fallecidos = serializers.IntegerField()


class TipoSiniestroStatSerializer(serializers.Serializer):
    """Para por_tipo_siniestro endpoint"""
    id = serializers.IntegerField()
    tipo = serializers.CharField()
    total_siniestros = serializers.IntegerField()
    total_lesionados = serializers.IntegerField()
    total_fallecidos = serializers.IntegerField()


class EvolucionAnualSiniestrosSerializer(serializers.Serializer):
    """Para evolucion_anual endpoint en Siniestros"""
    ano = serializers.IntegerField()
    total_siniestros = serializers.IntegerField()
    total_lesionados = serializers.IntegerField()
    total_fallecidos = serializers.IntegerField()


class EvolucionAnualVictimasSerializer(serializers.Serializer):

# 3. VIEWS.PY - Nuevos endpoints en ViewSets
# ============================================================================

# En SiniestroViewSet:
# ────────────────────

@action(detail=False, methods=['get'])
@method_decorator(cache_page(60 * 5))
def por_via(self, request):
    """GET /api/siniestros/por_via/?year=YYYY"""
    year = self._validate_year(request.query_params.get('year'))
    queryset = self.filter_queryset(self.get_queryset())
    stats = queryset.get_por_via(year=year)
    serializer = ViaStatSerializer(stats, many=True)
    return Response(serializer.data)


@action(detail=False, methods=['get'])
@method_decorator(cache_page(60 * 5))
def por_causa_probable(self, request):
    """GET /api/siniestros/por_causa_probable/?year=YYYY"""
    year = self._validate_year(request.query_params.get('year'))

    return Response(serializer.data)


@action(detail=False, methods=['get'])
@method_decorator(cache_page(60 * 5))

# BENEFICIOS DE LA ARQUITECTURA

"""
✅ Separación de Responsabilidades:
   - Managers: Lógica de agregación
   - Views: Orquestación de requests
   - Serializers: Estructuración de datos

✅ Reutilización:
   - Un método del manager puede usarse en múltiples vistas
   - Los queries pueden combinarse fácilmente

✅ Testing:
   - Fácil de testear cada capa independientemente
   - Tests unitarios en managers
   - Tests de integración en views

✅ Performance:
   - Caching automático con @method_decorator(cache_page())
   - Uso eficiente de ORM (bulk operations)
   - Índices en campos con `db_index=True`

✅ Documentación:
   - Automática con @extend_schema (Swagger/OpenAPI)
   - Self-documenting code
"""
